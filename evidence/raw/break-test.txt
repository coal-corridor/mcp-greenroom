## The script

#!/usr/bin/env bash
# Break test: run each command in the clone, print the command, its output and exit code.
set -u
clone="$1"; cd "$clone" || exit 2
port=8771
export MCP_GREENROOM_API="http://127.0.0.1:${port}"
data="$(mktemp -d)"
show() { printf '\n$ %s\n' "$*"; "$@" 2>&1; printf -- '-> exit %s\n' "$?"; }
serve() {  # serve ENV... : start the service in the background with extra env, wait for /livez
  env "$@" uv run --quiet uvicorn mcp_greenroom.api.main:app --host 127.0.0.1 --port "$port" --log-level warning >"$data/server.log" 2>&1 &
  pid=$!
  for _ in $(seq 1 60); do curl -fs "http://127.0.0.1:${port}/livez" >/dev/null 2>&1 && return 0; kill -0 $pid 2>/dev/null || return 1; sleep 0.5; done
  return 1
}
stop() { kill "$pid" 2>/dev/null; wait "$pid" 2>/dev/null; }

echo "### 0. Setup (the README has no setup section yet; CLAUDE.md commands used)"
show git log --oneline -1
show uv sync --quiet
show ./scripts/verify.sh quick

echo; echo "### 1. Missing config value"
show env MCP_GREENROOM_RISK_PROVIDER=live timeout 60 make run
show env MCP_GREENROOM_REJECT_THRESHOLD= timeout 60 make run
show env MCP_GREENROOM_REJECT_THRESHOLD=0.9 timeout 60 make run

echo; echo "### 2. Malformed input file (service running on a fresh store)"
db="sqlite:///${data}/greenroom.db"
serve MCP_GREENROOM_DATABASE_URL="$db" MCP_GREENROOM_SNAPSHOT_PATH="${data}/last-good.json" || { echo "service did not start"; cat "$data/server.log"; }
printf '{"server": {"name": "com.examplebank/x", ' > "$data/truncated.json"
printf '[1, 2, 3]\n' > "$data/array.json"
printf '{"server": {"name": "com.examplebank/x", "version": "1.0.0"}}\n' > "$data/no-tools.json"
show uv run --quiet mcp-greenroom submit "$data/truncated.json" --token dev-dana
show uv run --quiet mcp-greenroom submit "$data/array.json" --token dev-dana
show uv run --quiet mcp-greenroom submit "$data/no-tools.json" --token dev-dana
show uv run --quiet mcp-greenroom submit "$data/does-not-exist.json" --token dev-dana
show uv run --quiet mcp-greenroom submit sample-data/servers/payments-readonly.json --token dev-dana
stop
echo "--- service log (stderr) for section 2:"; cat "$data/server.log"

echo; echo "### 3. Store stopped"
echo "# 3a. Store unreachable from the start (database in a folder that does not exist)"
serve MCP_GREENROOM_DATABASE_URL="sqlite:///${data}/missing-dir/greenroom.db" MCP_GREENROOM_SNAPSHOT_PATH="${data}/none.json" || { echo "service did not start"; cat "$data/server.log"; }
show uv run --quiet mcp-greenroom health
show uv run --quiet mcp-greenroom list
show uv run --quiet mcp-greenroom submit sample-data/servers/payments-readonly.json --token dev-dana
stop
echo "--- service log (stderr) for 3a:"; cat "$data/server.log"
echo "# 3b. Store breaks while running (database file replaced after an approval)"
serve MCP_GREENROOM_DATABASE_URL="$db" MCP_GREENROOM_SNAPSHOT_PATH="${data}/last-good.json" || { echo "service did not start"; cat "$data/server.log"; }
printf 'this is not a database' > "${data}/greenroom.db"
show uv run --quiet mcp-greenroom list
show uv run --quiet mcp-greenroom status 00000000-0000-0000-0000-000000000000 --token dev-dana
show uv run --quiet mcp-greenroom submit sample-data/servers/ambiguous-exporter.json --token dev-dana
stop
echo "--- service log (stderr) for 3b:"; cat "$data/server.log"
rm -rf "$data"
